cmake_minimum_required(VERSION 3.21)
project(TrioBridgeRibbon LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_compile_options(/utf-8 /W4 /EHsc)
add_library(Ribbon SHARED Ribbon.cpp)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  target_sources(Ribbon PRIVATE Ribbon32.def)
else()
  target_sources(Ribbon PRIVATE Ribbon64.def)
endif()
target_link_libraries(Ribbon PRIVATE ole32 oleaut32 uuid)
add_executable(RibbonCheck RibbonCheck.cpp)
target_link_libraries(RibbonCheck PRIVATE ole32 oleaut32 uuid)
